API

H=https:///api
K=okl_key_...   # 开发者密钥
curl -H "Authorization: Bearer $K" $H/devices
curl -H "Authorization: Bearer $K" -X POST $H/devices/<id>/rpc -d '{"op":"status"}'
curl -H "Authorization: Bearer $K" -X PUT  $H/devices/<id>/images/<sha256> --data-binary @app.bin
curl -H "Authorization: Bearer $K" -X POST $H/devices/<id>/install -d '{"kind":"app","sha256":"<sha256>","version":"1.0"}'
curl -H "Authorization: Bearer $K" -X POST $H/devices/<id>/rpc -d '{"op":"py.exec","args":{"code":"print(1+1)"}}'
